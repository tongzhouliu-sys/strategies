3.4.4 每日强制剔除 50 只 — 两阶段算法 3.4.4 Daily mandatory eviction of 50 — Two-phase algorithm 
def daily_eviction_50():
    # 阶段1 / Phase 1: 规则触发强制剔除 → L4
    forced = check_all_dg_rules(pool)   # DG-IMM + DG-CUM triggered
    n_forced = len(forced)

    # 阶段2 / Phase 2: 补足50只 — 恶化分末位淘汰（仅L0/L1）
    remaining_quota = max(0, 50 - n_forced)
    if remaining_quota > 0:
        worst = sorted(pool_L0_L1, key=lambda x: x.deter_score)[: -remaining_quota]
        # 退回 UNTRACKED，不入黑名单
        move_back_to_untracked(worst)
